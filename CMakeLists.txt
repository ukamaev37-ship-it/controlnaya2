cmake_minimum_required(VERSION 3.16)
project(Control_Work_02_Polymorphism VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
if(MSVC)
  add_compile_options(/W4 /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
add_library(control_model uav.cpp sensor.cpp navigation.cpp controller.cpp autopilot.cpp)
target_include_directories(control_model PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(control_model PUBLIC Qt${QT_VERSION_MAJOR}::Core)
add_library(control_ui mainwindow.cpp labview.cpp)
target_include_directories(control_ui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(control_ui PUBLIC control_model Qt${QT_VERSION_MAJOR}::Widgets)
add_executable(control_work_02 main.cpp)
target_link_libraries(control_work_02 PRIVATE control_ui)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Test)
  add_executable(test_control_model tests/test_model.cpp)
  target_link_libraries(test_control_model PRIVATE control_model)
  add_test(NAME control_model COMMAND test_control_model)
  add_executable(test_control_window tests/test_window.cpp)
  target_link_libraries(test_control_window PRIVATE control_ui Qt${QT_VERSION_MAJOR}::Test)
  add_test(NAME control_window COMMAND test_control_window)
  set_tests_properties(control_window PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
